cmake_minimum_required(VERSION 3.16)
project(AoE2HDControl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(aoe2hd_world src/WorldReader.cpp src/WorldMonitor.cpp src/ResourceReader.cpp src/EconomyMonitor.cpp)
target_include_directories(aoe2hd_world PUBLIC include)

if(WIN32)
  add_executable(aoe2hd_probe src/main.cpp src/ProcessMemory.cpp)
  target_link_libraries(aoe2hd_probe PRIVATE aoe2hd_world)
  add_executable(aoe2hd_monitor src/monitor.cpp src/ProcessMemory.cpp)
  target_link_libraries(aoe2hd_monitor PRIVATE aoe2hd_world)
  add_executable(aoe2hd_economy_monitor src/economy_monitor.cpp src/ProcessMemory.cpp)
  target_link_libraries(aoe2hd_economy_monitor PRIVATE aoe2hd_world)
  if(MSVC)
    target_compile_options(aoe2hd_probe PRIVATE /W4)
    target_compile_options(aoe2hd_monitor PRIVATE /W4)
    target_compile_options(aoe2hd_economy_monitor PRIVATE /W4)
    target_compile_options(aoe2hd_world PRIVATE /W4)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(world_reader_tests tests/WorldReaderTests.cpp)
  target_link_libraries(world_reader_tests PRIVATE aoe2hd_world)
  add_test(NAME world_reader_tests COMMAND world_reader_tests)
  add_executable(world_monitor_tests tests/WorldMonitorTests.cpp)
  target_link_libraries(world_monitor_tests PRIVATE aoe2hd_world)
  add_test(NAME world_monitor_tests COMMAND world_monitor_tests)
  add_executable(resource_reader_tests tests/ResourceReaderTests.cpp)
  target_link_libraries(resource_reader_tests PRIVATE aoe2hd_world)
  add_test(NAME resource_reader_tests COMMAND resource_reader_tests)
  add_executable(economy_monitor_tests tests/EconomyMonitorTests.cpp)
  target_link_libraries(economy_monitor_tests PRIVATE aoe2hd_world)
  add_test(NAME economy_monitor_tests COMMAND economy_monitor_tests)
endif()

find_package(Python3 COMPONENTS Interpreter QUIET)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME economy_analyzer_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_analyze_economy.py)
  set_tests_properties(economy_analyzer_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/scripts)
endif()
