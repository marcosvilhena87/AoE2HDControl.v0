cmake_minimum_required(VERSION 3.16)
project(AoE2HDControl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(aoe2hd_world src/WorldReader.cpp src/WorldMonitor.cpp src/ResourceReader.cpp src/EconomyMonitor.cpp src/PopulationReader.cpp src/GameplayReadinessMonitor.cpp src/DerivedEventDetector.cpp src/UnitDiscovery.cpp src/UnitContentTracker.cpp src/UnitPointerProbe.cpp src/ObjectFieldSampler.cpp src/ProductionProgressProbe.cpp src/AbsoluteProgressMonitor.cpp)
target_include_directories(aoe2hd_world PUBLIC include)

if(WIN32)
  add_executable(aoe2hd_probe src/main.cpp src/ProcessMemory.cpp)
  target_link_libraries(aoe2hd_probe PRIVATE aoe2hd_world)
  add_executable(aoe2hd_monitor src/monitor.cpp src/ProcessMemory.cpp)
  target_link_libraries(aoe2hd_monitor PRIVATE aoe2hd_world)
  add_executable(aoe2hd_economy_monitor src/economy_monitor.cpp src/ProcessMemory.cpp)
  target_link_libraries(aoe2hd_economy_monitor PRIVATE aoe2hd_world)
  add_executable(aoe2hd_unit_discovery src/unit_discovery.cpp src/ProcessMemory.cpp)
  target_link_libraries(aoe2hd_unit_discovery PRIVATE aoe2hd_world)
  add_executable(aoe2hd_production_progress src/production_progress.cpp src/ProcessMemory.cpp)
  target_link_libraries(aoe2hd_production_progress PRIVATE aoe2hd_world)
  if(MSVC)
    target_compile_options(aoe2hd_probe PRIVATE /W4)
    target_compile_options(aoe2hd_monitor PRIVATE /W4)
    target_compile_options(aoe2hd_economy_monitor PRIVATE /W4)
    target_compile_options(aoe2hd_world PRIVATE /W4)
    target_compile_options(aoe2hd_unit_discovery PRIVATE /W4)
    target_compile_options(aoe2hd_production_progress PRIVATE /W4)
  endif()
endif()
